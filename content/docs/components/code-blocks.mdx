## File header

A `title` adds a header with the language and a copy button.

```ts title="src/shared/config/site.ts"
export const site = {
  name: "Docs",
  url: "http://localhost:3000",
};
```

## Plain block

Without a title the copy button floats in the corner and shows on hover or keyboard focus.

```ts
export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs));
}
```

## Highlighted lines

```tsx title="src/app/docs/layout.tsx"
export default function Layout({ children }: { children: ReactNode }) {
  const tree = source.getPageTree(); // [!code highlight]
  return <DocsLayout tree={tree}>{children}</DocsLayout>;
}
```

## Focus

Hover the block to bring the other lines back.

```ts
function getPage(slug: string[]) {
  const page = source.getPage(slug); // [!code focus]
  if (!page) notFound();
  return page;
}
```
